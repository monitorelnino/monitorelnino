    (function() {
        var seconds = 30;
        var targetUrl = 'https://servicos.to.gov.br/';
        var counter = document.getElementById('redirect-seconds');

        var interval = setInterval(function() {
            seconds -= 1;
            if (counter) {
                counter.textContent = seconds;
            }

                clearInterval(interval);

            }
        }, 1000);
