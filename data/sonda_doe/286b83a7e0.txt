        function adicionaEventoFiltroBuscaNorma() {
            if ($('#datatable_normas_filter').find("input[type='search']").is(':visible')) {
                var campoSearchFiltroNorma = $('#datatable_normas_filter').find("input[type='search']");
                
                campoSearchFiltroNorma.on('input', (function(){

                    for(var i = 0; i < normas.length; i++) {
                        var norma = $(normas[i]).parent().parent();
                        var textoNorma = norma.text().replace('Origem:', '').replace('Ementa:', '').replace(/Texto Integral:(.*)/, '').toLowerCase().replace(/[áàâã]/g,'a').replace(/[éèê]/g,'e').replace(/[óòôõ]/g,'o').replace(/[úùû]/g,'u').replace(/[ç]/g,'c').replace(/[íì]/g,'i');

                            norma.css('display', 'table-row');
